cmake_minimum_required(VERSION 3.16)
project(mohism_vins VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc calib3d video imgcodecs)
find_package(Threads REQUIRED)
set(MAVLINK_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/vendor/mavlink" CACHE PATH "MAVLink C v2 include root")
add_library(mohism_core src/vision.cpp src/filter.cpp src/config.cpp)
target_include_directories(mohism_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mohism_core PUBLIC ${OpenCV_LIBS})
target_compile_options(mohism_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mohism-vins src/main.cpp src/capture.cpp src/imu.cpp src/mavlink.cpp src/uvc.cpp)
target_include_directories(mohism-vins SYSTEM PRIVATE ${MAVLINK_INCLUDE_DIR})
target_link_libraries(mohism-vins PRIVATE mohism_core Threads::Threads)
target_compile_options(mohism-vins PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mohism-vins RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 add_executable(core-test tests/core_test.cpp)
 target_link_libraries(core-test PRIVATE mohism_core)
 add_test(NAME core COMMAND core-test)
endif()
