cmake_minimum_required(VERSION 3.16)
project(mohism_vins VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc calib3d video imgcodecs features2d)
find_package(Threads REQUIRED)
find_package(Ceres 2.1 REQUIRED)
set(MAVLINK_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/vendor/mavlink" CACHE PATH "MAVLink C v2 include root")
add_library(mohism_core STATIC src/vision.cpp src/feature_tracker.cpp src/preintegration.cpp src/estimator.cpp src/filter.cpp src/config.cpp src/tight_backend.cpp src/initialization.cpp src/marginal_prior.cpp src/loop_closure.cpp
 third_party/VINS-Fusion/ported/factor/pose_local_parameterization.cpp
 third_party/VINS-Fusion/ported/factor/projectionOneFrameTwoCamFactor.cpp
 third_party/VINS-Fusion/ported/factor/projectionTwoFrameOneCamFactor.cpp
 third_party/VINS-Fusion/ported/factor/projectionTwoFrameTwoCamFactor.cpp)
target_include_directories(mohism_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_include_directories(mohism_core SYSTEM PRIVATE third_party/VINS-Fusion/ported)
target_link_libraries(mohism_core PUBLIC ${OpenCV_LIBS} Ceres::ceres)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
 target_link_libraries(mohism_core PUBLIC stdc++fs)
endif()
target_compile_options(mohism_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math $<$<CONFIG:Release>:-O3>)
add_executable(mohism-vins src/main.cpp src/capture.cpp src/imu.cpp src/mavlink.cpp src/uvc.cpp)
target_include_directories(mohism-vins SYSTEM PRIVATE ${MAVLINK_INCLUDE_DIR})
target_link_libraries(mohism-vins PRIVATE mohism_core Threads::Threads)
target_compile_options(mohism-vins PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math $<$<CONFIG:Release>:-O3>)
install(TARGETS mohism-vins RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 add_executable(loop-test tests/loop_test.cpp)
 target_link_libraries(loop-test PRIVATE mohism_core)
 add_test(NAME loop COMMAND loop-test)
 add_executable(initialization-test tests/initialization_test.cpp)
 target_link_libraries(initialization-test PRIVATE mohism_core)
 add_test(NAME initialization COMMAND initialization-test)
 set_tests_properties(loop initialization PROPERTIES TIMEOUT 60)
 add_executable(tight-test tests/tight_test.cpp)
 target_include_directories(tight-test SYSTEM PRIVATE third_party/VINS-Fusion/ported)
 target_link_libraries(tight-test PRIVATE mohism_core)
 add_test(NAME tight COMMAND tight-test)
 set_tests_properties(tight PROPERTIES TIMEOUT 120)
 add_executable(core-test tests/core_test.cpp)
 target_link_libraries(core-test PRIVATE mohism_core)
 add_test(NAME core COMMAND core-test)
 add_executable(robustness-test tests/robustness_test.cpp)
 target_link_libraries(robustness-test PRIVATE mohism_core)
 add_test(NAME robustness COMMAND robustness-test)
 add_executable(mohism-replay host/replay.cpp)
 target_link_libraries(mohism-replay PRIVATE mohism_core)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME evaluation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/evaluation_test.py)
 endif()
 add_test(NAME replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/replay_test.py $<TARGET_FILE:mohism-replay>)
 add_executable(odometry-test tests/odometry_test.cpp src/mavlink.cpp)
 target_include_directories(odometry-test SYSTEM PRIVATE ${MAVLINK_INCLUDE_DIR})
 target_link_libraries(odometry-test PRIVATE mohism_core Threads::Threads util)
 add_test(NAME odometry COMMAND odometry-test)
 add_executable(trajectory-test tests/trajectory_test.cpp)
 target_link_libraries(trajectory-test PRIVATE mohism_core)
 add_test(NAME trajectory COMMAND trajectory-test)
 add_executable(uvc-test tests/uvc_test.cpp src/uvc.cpp)
 target_link_libraries(uvc-test PRIVATE mohism_core)
 add_test(NAME uvc COMMAND uvc-test)
 add_executable(profile-test tests/profile_test.cpp)
 target_link_libraries(profile-test PRIVATE mohism_core)
 add_test(NAME profile COMMAND profile-test)
 set_tests_properties(core robustness replay odometry trajectory profile PROPERTIES TIMEOUT 30)
endif()
